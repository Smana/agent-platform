# Portable run runtime (adoption 3a) Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Run an agent on any Kubernetes cluster with Cilium and gVisor, without Crossplane: a native `agents.ogenki.io/v1alpha1 AgentRun` reconciled by a new `run-controller` binary, read by the broker through a configurable `runwatch` seam.

**Architecture:**

- **Seam first (Tasks 1–2):** `runwatch`'s three constants (GVK, namespace, name prefix) become one `runwatch.Kind`, set once at startup by `runwatch.Configure` from the broker's and the factory's config. Its default is today's value, so cloud-native-ref is untouched.
- **API (Task 3):** a typed `AgentRun` in `api/v1alpha1`, generated by controller-gen. Its spec copies the Crossplane XRD's schema and CEL rules.
- **Controller (Tasks 4–8):** `internal/runctl` holds the config, pure rendering of the five owned objects (rules, ConfigMaps, ServiceAccount, CNP, Pod), a pure status function ported from the composition's `_phaseOf`/`_reasonOf`, and a controller-runtime reconciler. `cmd/run-controller` wires it, with an image and a release entry.
- **Harness (Task 9):** two file modes, one for the model key and one for the git token, and an optional `MCP_URL`.

**Tech Stack:** Go 1.27, controller-runtime v0.25.2, controller-gen v0.22, envtest 1.36.x, Cilium `cilium.io/v2` (unstructured), Python 3 stdlib `unittest` for the harness.

**Spec:** [`docs/superpowers/specs/2026-10-18-portable-run-runtime-design.md`](../specs/2026-10-18-portable-run-runtime-design.md). Read it first. The production shape being ported is crossplane-configuration v0.9.15, `apis/agentrun/kcl/main.k` and `apis/agentrun/definition.yaml`. Open them beside Tasks 3–6.

**Provenance:** the code below was written against `origin/main` at `4f071d5` but has **not** been compiled. Fix compile details (imports, a renamed helper) as you go. Do not change behaviour without updating this plan.

```go
// SPDX-License-Identifier: Apache-2.0

package runwatch

import (
	"errors"
	"fmt"
	"regexp"
	"sync/atomic"

	"k8s.io/apimachinery/pkg/runtime/schema"
)

// Kind is where AgentRuns live and how they are named: cloud-native-ref's Crossplane XR by
// default, agent-platform's own CRD (run-controller, adoption 3a) when configured.
type Kind struct {
	Group      string `json:"group,omitempty"`
	Namespace  string `json:"namespace,omitempty"`
	NamePrefix string `json:"namePrefix,omitempty"`
}

// DefaultKind is cloud-native-ref's: SP1's XRD, which names a run xplane-run-<C2 id> (C2).
var DefaultKind = Kind{Group: "cloud.ogenki.io", Namespace: "agents", NamePrefix: "xplane-run-"}

var (
	groupRE  = regexp.MustCompile(`^[a-z0-9]([a-z0-9-]*[a-z0-9])?(\.[a-z0-9]([a-z0-9-]*[a-z0-9])?)+$`)
	nsRE     = regexp.MustCompile(`^[a-z0-9]([a-z0-9-]{0,61}[a-z0-9])?$`)
	prefixRE = regexp.MustCompile(`^[a-z0-9][a-z0-9-]{0,40}-$`) // + 8 id characters stays a DNS label
)

type state struct {
	kind    Kind
	subject *regexp.Regexp
}

// current is read on every claim; atomic so a test's Configure never races a watch goroutine.
var current atomic.Pointer[state]

func init() { current.Store(newState(DefaultKind)) }

func newState(k Kind) *state {
	return &state{kind: k, subject: regexp.MustCompile("^system:serviceaccount:" +
		regexp.QuoteMeta(k.Namespace) + ":" + regexp.QuoteMeta(k.NamePrefix) + "([a-z2-7]{8})$")}
}

// WithDefaults fills each empty field from DefaultKind.
func (k Kind) WithDefaults() Kind {
	if k.Group == "" {
		k.Group = DefaultKind.Group
	}
	if k.Namespace == "" {
		k.Namespace = DefaultKind.Namespace
	}
	if k.NamePrefix == "" {
		k.NamePrefix = DefaultKind.NamePrefix
	}
	return k
}

// Validate reports every malformed field at once.
func (k Kind) Validate() error {
	var errs []error
	if !groupRE.MatchString(k.Group) {
		errs = append(errs, fmt.Errorf("group %q is not an API group (a DNS subdomain with a dot)", k.Group))
	}
	if !nsRE.MatchString(k.Namespace) {
		errs = append(errs, fmt.Errorf("namespace %q is not a DNS label", k.Namespace))
	}
	if !prefixRE.MatchString(k.NamePrefix) {
		errs = append(errs, fmt.Errorf("namePrefix %q must match %s", k.NamePrefix, prefixRE))
	}
	return errors.Join(errs...)
}

// Configure sets the kind every reader uses, its empty fields defaulted. Call it once at
// startup, before a watch or a client reads a run. A refused kind leaves the current one.
func Configure(k Kind) error {
	k = k.WithDefaults()
	if err := k.Validate(); err != nil {
		return err
	}
	current.Store(newState(k))
	return nil
}

// Current is the configured kind.
func Current() Kind { return current.Load().kind }

// Namespace is where every run lives.
func Namespace() string { return Current().Namespace }

// ClaimPrefix starts every run's name: <prefix><C2 id>.
func ClaimPrefix() string { return Current().NamePrefix }

// GVK is the AgentRun's kind (C3).
func GVK() schema.GroupVersionKind {
	return schema.GroupVersionKind{Group: Current().Group, Version: "v1alpha1", Kind: "AgentRun"}
}

// RunSubject matches a run's ServiceAccount token subject, capturing the run id: the
// ServiceAccount is named exactly like the run, on both runtimes.
func RunSubject() *regexp.Regexp { return current.Load().subject }
```

```go
// Package runctl is the run-controller (adoption 3a): it reconciles agents.ogenki.io AgentRuns
// into the pod, ServiceAccount, ConfigMaps and CiliumNetworkPolicy that cloud-native-ref's
// Crossplane composition builds, minus the gateway, octo-sts, traces, Kueue and suspend.
package runctl

import (
	"bytes"
	"encoding/json"
	"errors"
	"fmt"
	"net/url"
	"os"
	"path/filepath"
	"regexp"

	"github.com/Smana/agent-platform/internal/envelope"

// Config is the run-controller's file, from the chart.
type Config struct {
	Namespace        string `json:"namespace"`
	NamePrefix       string `json:"namePrefix"`
	RuntimeClassName string `json:"runtimeClassName,omitempty"`
	// AllowUnsandboxed places pods without a RuntimeClass; each run then carries condition
	// Unsandboxed, so the room can say so (3a R5).
	AllowUnsandboxed bool                `json:"allowUnsandboxed,omitempty"`
	ModelSecret      string              `json:"modelSecret"`
	GitSecret        string              `json:"gitSecret"`
	Broker           Broker              `json:"broker"`
	Images           Images              `json:"images"`
	EgressProfiles   map[string][]string `json:"egressProfiles,omitempty"`
}

// Broker is where the bridge connects, and what the CNP lets it reach.
type Broker struct {
	URL       string            `json:"url"`
	Namespace string            `json:"namespace"`
	Labels    map[string]string `json:"labels"`
	CASecret  string            `json:"caSecret"`
}

// Images are the run pod's two images, pinned by the chart.
type Images struct {
	Harness string `json:"harness"`
	Bridge  string `json:"bridge"`
}

// defaultProfiles are the composition's _EGRESS_PROFILES (crossplane-configuration v0.9.15).
var defaultProfiles = map[string][]string{
	"github": {"github.com", "api.github.com", "codeload.github.com", "objects.githubusercontent.com", "raw.githubusercontent.com"},
	"pypi":   {"pypi.org", "files.pythonhosted.org"},
	"npm":    {"registry.npmjs.org"},
	"golang": {"proxy.golang.org", "sum.golang.org"},
	"crates": {"index.crates.io", "static.crates.io"},
}

var prefixRE = regexp.MustCompile(`^[a-z0-9][a-z0-9-]{0,40}-$`)

// LoadConfig reads, strictly decodes, defaults and validates the file at path.
func LoadConfig(path string) (Config, error) {
	var c Config
	raw, err := os.ReadFile(filepath.Clean(path))
	if err != nil {
		return c, fmt.Errorf("config: %w", err)
	}
	j, err := yaml.YAMLToJSON(raw)
	if err != nil {
		return c, fmt.Errorf("config %s: %w", path, err)
	}
	dec := json.NewDecoder(bytes.NewReader(j))
	dec.DisallowUnknownFields()
	if err := dec.Decode(&c); err != nil {
		return c, fmt.Errorf("config %s: %w", path, err)
	}
	if c.EgressProfiles == nil {
		c.EgressProfiles = defaultProfiles
	}
	if err := c.validate(); err != nil {
		return c, fmt.Errorf("config %s: %w", path, err)
	}
	return c, nil
}

func (c Config) validate() error {
	var errs []error
	req := func(name, v string) {
		if v == "" {
			errs = append(errs, fmt.Errorf("%s is required", name))
		}
	}
	req("namespace", c.Namespace)
	req("modelSecret", c.ModelSecret)
	req("gitSecret", c.GitSecret)
	req("broker.namespace", c.Broker.Namespace)
	req("broker.caSecret", c.Broker.CASecret)
	req("images.harness", c.Images.Harness)
	req("images.bridge", c.Images.Bridge)
	if !prefixRE.MatchString(c.NamePrefix) {
		errs = append(errs, fmt.Errorf("namePrefix %q must match %s", c.NamePrefix, prefixRE))
	}
	if u, err := url.Parse(c.Broker.URL); err != nil || u.Scheme != "https" || u.Hostname() == "" {
		errs = append(errs, fmt.Errorf("broker.url %q must be an https URL", c.Broker.URL))
	}
	if len(c.Broker.Labels) == 0 {
		errs = append(errs, errors.New("broker.labels: the CNP needs at least one to select the broker"))
	}
	if len(c.EgressProfiles["github"]) == 0 {
		errs = append(errs, errors.New("egressProfiles.github is required: every run clones from GitHub"))
	}
	return errors.Join(errs...)
}

// RunID is the run's C2 id, or "" for a name this controller does not own.
func (c Config) RunID(r *v1alpha1.AgentRun) string {
	id, ok := strings.CutPrefix(r.Name, c.NamePrefix)
	if !ok || !envelope.ValidID(id) {
		return ""
	}
	return id
}

// Invalid says why the controller refuses r, or "" when it accepts it (3a R6, R7).
func (c Config) Invalid(r *v1alpha1.AgentRun) string {
	switch {
	case c.RunID(r) == "":
		return fmt.Sprintf("the name must be namePrefix %q followed by 8 characters of [a-z2-7]", c.NamePrefix)
	case r.Namespace != c.Namespace:
		return fmt.Sprintf("runs live in namespace %q", c.Namespace)
	case r.Spec.Suspendable:
		return "suspendable runs need cloud-native-ref's runtime"
	case r.Spec.DataClass == "internal":
		return "internal runs need the model gateway, which this runtime does not have"
	case c.RuntimeClassName == "" && !c.AllowUnsandboxed:
		return "runtimeClassName is empty and allowUnsandboxed is false: refusing to run unsandboxed"
	}
	for _, p := range r.Spec.Egress.Profiles {
		if _, ok := c.EgressProfiles[string(p)]; !ok {
			return fmt.Sprintf("egress profile %q is not configured", p)
		}
	}
	return ""
}

// FQDNs are the hosts the run may reach on 443: github, then its profiles, deduplicated.
func (c Config) FQDNs(r *v1alpha1.AgentRun) []string {
	out := slices.Clone(c.EgressProfiles["github"])
	for _, p := range r.Spec.Egress.Profiles {
		for _, h := range c.EgressProfiles[string(p)] {
			if !slices.Contains(out, h) {
				out = append(out, h)
			}
		}
	}
	return out
}

// Model is the provider the harness calls directly (3a R4), from modelSecret.
type Model struct {
	BaseURL, Name, Host string
	Port                int
}

const defaultBaseURL = "https://api.anthropic.com/v1"

// ModelFrom reads modelSecret's data: api_key and model required, base_url defaulted.
func ModelFrom(data map[string][]byte) (Model, error) {
	if len(data["api_key"]) == 0 {
		return Model{}, errors.New("modelSecret has no api_key")
	}
	m := Model{BaseURL: strings.TrimSpace(string(data["base_url"])), Name: strings.TrimSpace(string(data["model"]))}
	if m.Name == "" {
		return Model{}, errors.New("modelSecret has no model")
	}
	if m.BaseURL == "" {
		m.BaseURL = defaultBaseURL
	}
	u, err := url.Parse(m.BaseURL)
	if err != nil || (u.Scheme != "https" && u.Scheme != "http") || u.Hostname() == "" {
		return Model{}, fmt.Errorf("modelSecret base_url %q is not an http(s) URL", m.BaseURL)
	}
	m.Host, m.Port = u.Hostname(), 443
	if u.Scheme == "http" {
		m.Port = 80
	}
	if p := u.Port(); p != "" {
		m.Port, _ = strconv.Atoi(p)
	}
	return m, nil
}
```

```go
// NetworkPolicy is the run's default-deny CNP: DNS for its names only, the broker on 8443,
// github and its profiles on 443, the model host on its port. Nothing dials in except the
// kubelet's probe of the bridge (C4).
func NetworkPolicy(r *v1alpha1.AgentRun, c Config, m Model, runID string) *unstructured.Unstructured {
	web := c.FQDNs(r) // reached on 443
	if m.Port == 443 && !slices.Contains(web, m.Host) {
		web = append(web, m.Host)
	}
	names := slices.Clone(web)
	if !slices.Contains(names, m.Host) {
		names = append(names, m.Host)
	}
	if r.Spec.RoomRef != "" {
		names = append(names, brokerHost(c))
	}
	matchNames := func(hs []string) []any {
		out := make([]any, 0, len(hs))
		for _, h := range hs {
			out = append(out, map[string]any{"matchName": h})
		}
		return out
	}
	port := func(p int) []any {
		return []any{map[string]any{"ports": []any{map[string]any{"port": strconv.Itoa(p), "protocol": "TCP"}}}}
	}
	egress := []any{
		map[string]any{
			"toEndpoints": []any{map[string]any{"matchLabels": map[string]any{"io.kubernetes.pod.namespace": "kube-system", "k8s-app": "kube-dns"}}},
			"toPorts": []any{map[string]any{
				"ports": []any{map[string]any{"port": "53", "protocol": "UDP"}, map[string]any{"port": "53", "protocol": "TCP"}},
				"rules": map[string]any{"dns": matchNames(names)},
			}},
		},
		map[string]any{"toFQDNs": matchNames(web), "toPorts": port(443)},
	}
	if m.Port != 443 {
		egress = append(egress, map[string]any{"toFQDNs": matchNames([]string{m.Host}), "toPorts": port(m.Port)})
	}
	var ingress []any
	if r.Spec.RoomRef != "" {
		sel := map[string]any{"io.kubernetes.pod.namespace": c.Broker.Namespace}
		for k, v := range c.Broker.Labels {
			sel[k] = v
		}
		egress = append(egress, map[string]any{"toEndpoints": []any{map[string]any{"matchLabels": sel}}, "toPorts": port(8443)})
		ingress = []any{map[string]any{"fromEntities": []any{"host"}, "toPorts": port(BridgeHealthPort)}}
	}
	spec := map[string]any{
		"endpointSelector": map[string]any{"matchLabels": map[string]any{LabelRunID: runID}},
		"egress":           egress,
	}
	// An empty ingress list still makes the endpoint default-deny for ingress in Cilium.
	spec["ingress"] = ingress
	if ingress == nil {
		spec["ingress"] = []any{}
	}
	u := &unstructured.Unstructured{Object: map[string]any{"apiVersion": "cilium.io/v2", "kind": "CiliumNetworkPolicy", "spec": spec}}
	om := meta(r, runID, r.Name)
	u.SetName(om.Name)
	u.SetNamespace(om.Namespace)
	u.SetLabels(om.Labels)
	return u
}
```

```go
// SPDX-License-Identifier: Apache-2.0

package runctl

import (
	"slices"
	"time"

	corev1 "k8s.io/api/core/v1"

	v1alpha1 "github.com/Smana/agent-platform/api/v1alpha1"
	"github.com/Smana/agent-platform/internal/runwatch"
)

// The composition's revoke values: a budget one ends the run BudgetExhausted, manual ends it
// Revoked; anything else is ignored (crossplane-configuration v0.9.15, _phaseOf).
var budgetReasons = []string{"budget-run", "budget-principal", "budget-fleet"}

// RevokeEnds reports whether the revoke annotation's value ends the run.
func RevokeEnds(revoked string) bool { return revoked == "manual" || slices.Contains(budgetReasons, revoked) }

// Observe is the composition's _phaseOf and _reasonOf over a plain pod: a terminal status is
// final; then revocation; then the harness's exit; then the pod's failure; then a pod lost
// after the run started. finishedAt is always set at a terminal phase, so runwatch never falls
// back to the Ready condition.
func Observe(prev v1alpha1.AgentRunStatus, revoked string, pod *corev1.Pod, podGone bool, now time.Time) v1alpha1.AgentRunStatus {
	if runwatch.Terminal(prev.Phase) {
		return prev
	}
	st := v1alpha1.AgentRunStatus{Phase: "Pending", StartedAt: prev.StartedAt}
	if pod != nil && st.StartedAt == "" && pod.Status.Phase != corev1.PodPending {
		st.StartedAt = pod.CreationTimestamp.UTC().Format(time.RFC3339)
	}
	end := func(phase, reason string, at time.Time) v1alpha1.AgentRunStatus {
		st.Phase, st.Reason, st.FinishedAt = phase, reason, at.UTC().Format(time.RFC3339)
		return st
	}
	switch {
	case slices.Contains(budgetReasons, revoked):
		return end("BudgetExhausted", revoked, now)
	case revoked == "manual":
		return end("Revoked", revoked, now)
	}
	started := prev.Phase == "Running"
	if pod == nil {
		if started || podGone && prev.StartedAt != "" {
			return end("Failed", "PodLost", now)
		}
		return st
	}
	if t := harnessTerminated(pod); t != nil {
		if t.ExitCode == 0 {
			return end("Succeeded", "", t.FinishedAt.Time)
		}
		return end("Failed", "PodFailed", t.FinishedAt.Time)
	}
	if pod.Status.Phase == corev1.PodFailed {
		if podCondition(pod, corev1.DisruptionTarget) {
			return end("Failed", "Disrupted", now)
		}
		return end("Failed", "PodFailed", now)
	}
	if pod.DeletionTimestamp != nil && started {
		return end("Failed", "PodLost", now)
	}
	if started || podCondition(pod, corev1.PodReady) {
		st.Phase = "Running"
	}
	return st
}

func harnessTerminated(p *corev1.Pod) *corev1.ContainerStateTerminated {
	for _, s := range p.Status.ContainerStatuses {
		if s.Name == "harness" && s.State.Terminated != nil {
			return s.State.Terminated
		}
	}
	return nil
}

func podCondition(p *corev1.Pod, t corev1.PodConditionType) bool {
	for _, c := range p.Status.Conditions {
		if c.Type == t {
			return c.Status == corev1.ConditionTrue
		}
	}
	return false
}
```

```go
package runctl

import (
	"context"
```

```go
const secretRetry = 30 * time.Second

// Reconciler turns an AgentRun into its owned objects and reports the pod back as status.
// Owned objects are created once: a run's spec is immutable, so there is nothing to patch.
type Reconciler struct {
	Client  client.Client
	Secrets client.Reader // uncached: the controller never lists or watches Secrets
	Cfg     Config
	Now     func() time.Time
	Log     *slog.Logger
}

// SetupWithManager watches runs and the pods they own.
func (r *Reconciler) SetupWithManager(mgr ctrl.Manager) error {
	return ctrl.NewControllerManagedBy(mgr).Named("run-controller").
		For(&v1alpha1.AgentRun{}).Owns(&corev1.Pod{}).Complete(r)
}

// Reconcile is idempotent: a restart reconciles every non-terminal run from scratch.
func (r *Reconciler) Reconcile(ctx context.Context, req ctrl.Request) (ctrl.Result, error) {
	run := &v1alpha1.AgentRun{}
	if err := r.Client.Get(ctx, req.NamespacedName, run); err != nil {
		return ctrl.Result{}, client.IgnoreNotFound(err)
	}
	if runwatch.Terminal(run.Status.Phase) {
		return ctrl.Result{}, nil
	}
	now := r.Now()
	st := run.Status.DeepCopy()
	if why := r.Cfg.Invalid(run); why != "" {
		st.Phase, st.FinishedAt = "Failed", now.UTC().Format(time.RFC3339)
		meta.SetStatusCondition(&st.Conditions, metav1.Condition{Type: "Invalid", Status: metav1.ConditionTrue, Reason: "Refused", Message: why})
		return ctrl.Result{}, r.writeStatus(ctx, run, *st)
	}
	runID := r.Cfg.RunID(run)
	st.RunID, st.Branch, st.ConversationID = runID, Branch(run, runID), string(run.UID)
	revoked := run.Annotations[runwatch.RevokedAnnotation]

	pod, gone, err := r.pod(ctx, run)
	if err != nil {
		return ctrl.Result{}, err
	}
	if RevokeEnds(revoked) {
		if pod != nil && pod.DeletionTimestamp == nil {
			if err := r.Client.Delete(ctx, pod); client.IgnoreNotFound(err) != nil {
				return ctrl.Result{}, err
			}
		}
		return ctrl.Result{}, r.writeStatus(ctx, run, merge(*st, Observe(run.Status, revoked, pod, gone, now)))
	}
	if pod == nil && !gone {
		m, ca, missing, err := r.inputs(ctx)
		if err != nil {
			return ctrl.Result{}, err
		}
		if missing != "" {
			meta.SetStatusCondition(&st.Conditions, metav1.Condition{Type: "Ready", Status: metav1.ConditionFalse, Reason: "SecretMissing", Message: missing})
			if st.Phase == "" {
				st.Phase = "Pending"
			}
			return ctrl.Result{RequeueAfter: secretRetry}, r.writeStatus(ctx, run, *st)
		}
		if err := r.create(ctx, run, runID, m, ca); err != nil {
			return ctrl.Result{}, err
		}
		meta.SetStatusCondition(&st.Conditions, metav1.Condition{Type: "Ready", Status: metav1.ConditionTrue, Reason: "Created"})
		if r.Cfg.RuntimeClassName == "" {
			meta.SetStatusCondition(&st.Conditions, metav1.Condition{Type: "Unsandboxed", Status: metav1.ConditionTrue, Reason: "AllowUnsandboxed",
				Message: "the pod runs without a RuntimeClass: allowUnsandboxed is set"})
		}
	}
	return ctrl.Result{}, r.writeStatus(ctx, run, merge(*st, Observe(run.Status, revoked, pod, gone, now)))
}

// pod is the run's pod; gone is that it existed (the run has a startedAt) and does not now.
func (r *Reconciler) pod(ctx context.Context, run *v1alpha1.AgentRun) (*corev1.Pod, bool, error) {
	p := &corev1.Pod{}
	err := r.Client.Get(ctx, types.NamespacedName{Namespace: run.Namespace, Name: run.Name}, p)
	if apierrors.IsNotFound(err) {
		return nil, run.Status.StartedAt != "", nil
	}
	if err != nil {
		return nil, false, err
	}
	return p, false, nil
}

// inputs reads the model Secret, the git Secret's presence and the broker CA. missing names
// the first absent one: the chart may create them after the controller starts.
func (r *Reconciler) inputs(ctx context.Context) (Model, []byte, string, error) {
	get := func(ns, name string) (*corev1.Secret, error) {
		s := &corev1.Secret{}
		return s, r.Secrets.Get(ctx, types.NamespacedName{Namespace: ns, Name: name}, s)
	}
	ms, err := get(r.Cfg.Namespace, r.Cfg.ModelSecret)
	if apierrors.IsNotFound(err) {
		return Model{}, nil, fmt.Sprintf("Secret %s/%s not found", r.Cfg.Namespace, r.Cfg.ModelSecret), nil
	} else if err != nil {
		return Model{}, nil, "", err
	}
	m, err := ModelFrom(ms.Data)
	if err != nil {
		return Model{}, nil, err.Error(), nil
	}
	if gs, err := get(r.Cfg.Namespace, r.Cfg.GitSecret); apierrors.IsNotFound(err) || err == nil && len(gs.Data["token"]) == 0 {
		return Model{}, nil, fmt.Sprintf("Secret %s/%s has no token", r.Cfg.Namespace, r.Cfg.GitSecret), nil
	} else if err != nil {
		return Model{}, nil, "", err
	}
	cs, err := get(r.Cfg.Broker.Namespace, r.Cfg.Broker.CASecret)
	if apierrors.IsNotFound(err) || err == nil && len(cs.Data["ca.crt"]) == 0 {
		return Model{}, nil, fmt.Sprintf("Secret %s/%s has no ca.crt", r.Cfg.Broker.Namespace, r.Cfg.Broker.CASecret), nil
	} else if err != nil {
		return Model{}, nil, "", err
	}
	return m, cs.Data["ca.crt"], "", nil
}

// create makes every owned object, the pod last so its volumes and ServiceAccount exist.
func (r *Reconciler) create(ctx context.Context, run *v1alpha1.AgentRun, runID string, m Model, ca []byte) error {
	task, err := TaskConfigMap(run, runID)
	if err != nil {
		return err
	}
	for _, o := range []client.Object{ServiceAccount(run), task, CAConfigMap(run, ca),
		NetworkPolicy(run, r.Cfg, m, runID), Pod(run, r.Cfg, m, runID)} {
		if err := controllerutil.SetControllerReference(run, o, r.Client.Scheme()); err != nil {
			return err
		}
		if err := r.Client.Create(ctx, o); err != nil && !apierrors.IsAlreadyExists(err) {
			return fmt.Errorf("create %T %s: %w", o, o.GetName(), err)
		}
	}
	return nil
}

// merge lays Observe's phase, reason and times over the reconciler's identity fields and conditions.
func merge(base, obs v1alpha1.AgentRunStatus) v1alpha1.AgentRunStatus {
	base.Phase, base.Reason, base.StartedAt, base.FinishedAt = obs.Phase, obs.Reason, obs.StartedAt, obs.FinishedAt
	return base
}

func (r *Reconciler) writeStatus(ctx context.Context, run *v1alpha1.AgentRun, st v1alpha1.AgentRunStatus) error {
	if equality.Semantic.DeepEqual(run.Status, st) {
		return nil
	}
	run.Status = st
	return r.Client.Status().Update(ctx, run)
}
```

```go
// RunController runs the run-controller (adoption 3a) until ctx ends. Its cache holds runs and
// pods in the runs' namespace only; Secrets are read uncached, so it never lists them.
func RunController(ctx context.Context, logger *slog.Logger, getenv func(string) string) error {
	path := getenv("RUN_CONTROLLER_CONFIG")
	if path == "" {
		path = defaultRunControllerConfig
	}
	cfg, err := runctl.LoadConfig(path)
	if err != nil {
		return err
	}
	lr := logr.FromSlogHandler(logger.Handler()) // as the broker and the factory do
	ctrl.SetLogger(lr)
	klog.SetLogger(lr)
	s := runtime.NewScheme()
	if err := clientgoscheme.AddToScheme(s); err != nil {
		return err
	}
	if err := v1alpha1.AddToScheme(s); err != nil {
		return err
	}
	ns := map[string]cache.Config{cfg.Namespace: {}}
	drain := 30 * time.Second
	mgr, err := ctrl.NewManager(ctrl.GetConfigOrDie(), ctrl.Options{
		Scheme: s, LeaderElection: true, LeaderElectionID: "run-controller", LeaderElectionNamespace: cfg.Namespace,
		LeaderElectionReleaseOnCancel: true, GracefulShutdownTimeout: &drain,
		Metrics: metricsserver.Options{BindAddress: ":9090"}, HealthProbeBindAddress: ":9091",
		Cache: cache.Options{ByObject: map[client.Object]cache.ByObject{
			&v1alpha1.AgentRun{}: {Namespaces: ns}, &corev1.Pod{}: {Namespaces: ns},
		}},
	})
	if err != nil {
		return fmt.Errorf("run-controller: manager: %w", err)
	}
	if err := mgr.AddHealthzCheck("ping", healthz.Ping); err != nil {
		return err
	}
	if err := mgr.AddReadyzCheck("ping", healthz.Ping); err != nil {
		return err
	}
	r := &runctl.Reconciler{Client: mgr.GetClient(), Secrets: mgr.GetAPIReader(), Cfg: cfg, Now: time.Now, Log: logger}
	if err := r.SetupWithManager(mgr); err != nil {
		return err
	}
	logger.Info("run-controller starting", "namespace", cfg.Namespace, "prefix", cfg.NamePrefix, "runtimeClass", cfg.RuntimeClassName)
	return mgr.Start(ctx)
}
```
